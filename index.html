Cakeflakes_Orders Apps Script 
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Cakeflakes_Orders') || ss.getSheets()[0];
  const tz = ss.getSpreadsheetTimeZone();
  const fmtDate = (v) => v instanceof Date ? Utilities.formatDate(v, tz, 'yyyy-MM-dd') : String(v || '').trim();
  const rows = sheet.getDataRange().getValues();
  const orders = rows.slice(1).filter(r => r[2]).map((r, i) => ({
    id: String(r[0] || `CF26${String(i + 1).padStart(2, '0')}`).trim(),
    pickupDate: fmtDate(r[1]),
    customerName: String(r[2] || '').trim(),
    phone: String(r[3] || '').trim(),
    eventType: String(r[4] || 'Birthday').trim(),
    themeTitle: String(r[5] || '').trim(),
    tiers: String(r[6] || '1 Tier').trim(),
    tierSpecs: String(r[7] || '').trim(),
    frostingColors: String(r[8] || '').trim(),
    notes: String(r[9] || '').trim(),
    totalPrice: parseFloat(r[10]) || 0,
    advancePaid: parseFloat(r[11]) || 0,
    advanceMode: String(r[12] || 'Zelle').trim(),
    finalPaid: parseFloat(r[13]) || 0,
    phase: String(r[14] || 'Order Received').trim()
  }));
  const json = JSON.stringify({ orders });
  const cb = e && e.parameter && e.parameter.callback;
  return cb
    ? ContentService.createTextOutput(`${cb}(${json})`).setMimeType(ContentService.MimeType.JAVASCRIPT)
    : ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}

Bank reconcilation Apps Script (Cols A–J, Row 5+) 
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('All Transactions') || ss.getSheets()[0];
  const tz = ss.getSpreadsheetTimeZone();
  const fmtDate = (v) => v instanceof Date ? Utilities.formatDate(v, tz, 'yyyy-MM-dd') : String(v || '').trim();
  const rows = sheet.getDataRange().getValues();
  const transactions = rows.slice(4)
    .filter(r => r[6] && !String(r[0]).toLowerCase().includes('total') && !String(r[6]).toLowerCase().includes('total'))
    .map((r, i) => ({
      txNum: String(r[0] || (i + 1)).trim(),
      date: fmtDate(r[1]),
      year: String(r[2] || '').trim(),
      statementMonth: String(r[3] || '').trim(),
      section: String(r[4] || '').trim(),
      category: String(r[5] || '').trim(),
      party: String(r[6] || '').trim(),
      deposit: parseFloat(r[7]) || 0,
      withdrawal: parseFloat(r[8]) || 0,
      netImpact: parseFloat(r[9]) || 0
    }));
  const json = JSON.stringify({ transactions });
  const cb = e && e.parameter && e.parameter.callback;
  return cb
    ? ContentService.createTextOutput(`${cb}(${json})`).setMimeType(ContentService.MimeType.JAVASCRIPT)
    : ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
Cakeflakes Amazon Expenses & Returns 2026

Connect "Amazon Net Orders 2026" Sheet

Reads the last sheet Amazon Net Orders 2026 (Columns A–L, starting Row 2) without affecting your Orders or Bank sheets.

Read-Only Script for "Amazon Net Orders 2026" (Cols A–L, Row 2+)
  1. Open your Cakeflakes Amazon Expenses & Returns 2026 Google Sheet → Extensions → Apps Script.
  2. Paste the script below, save, and click Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone).
  3. Paste the Web App URL into the box above and click Save & Load Amazon Net Orders.
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheets = ss.getSheets();
  // Targets "Amazon Net Orders 2026" or the last tab in the workbook
  const sheet = ss.getSheetByName('Amazon Net Orders 2026') || sheets[sheets.length - 1];
  const tz = ss.getSpreadsheetTimeZone();
  const fmtDate = (val) => (val instanceof Date)
    ? Utilities.formatDate(val, tz, 'yyyy-MM-dd')
    : String(val || '').trim();

  // Header is Row 1; data starts at Row 2 (Cols A through L)
  const rows = sheet.getDataRange().getValues();
  const amazonOrders = rows.slice(1)
    .filter(r => r[1] && String(r[1]).trim() !== '' && !String(r[0]).toLowerCase().includes('total'))
    .map((r, i) => ({
      orderDate: fmtDate(r[0]),
      orderId: String(r[1] || `AMZ-${i + 1}`).trim(),
      title: String(r[2] || '').trim(),
      category: String(r[3] || 'Amazon Supplies').trim(),
      unitPrice: parseFloat(r[4]) || 0,
      orderedQty: parseFloat(r[5]) || 1,
      returnedQty: parseFloat(r[6]) || 0,
      netQty: parseFloat(r[7]) || 1,
      grossSpend: parseFloat(r[8]) || 0,
      refundAmount: parseFloat(r[9]) || 0,
      netExpense: parseFloat(r[10]) || ((parseFloat(r[8]) || 0) - (parseFloat(r[9]) || 0)),
      status: String(r[11] || 'Netted').trim()
    }));

  const jsonStr = JSON.stringify({ amazonOrders });
  const cb = e && e.parameter && e.parameter.callback;
  if (cb) {
    return ContentService.createTextOutput(`${cb}(${jsonStr})`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonStr)
    .setMimeType(ContentService.MimeType.JSON);
}
