Google Sheet Master Workbook Sync

1-Click Clean 15-Column Table Builder (Cakeflakes_Orders)
  1. First-Time Setup Only: Open your Google Sheet → Extensions → Apps Script, paste this script, select setupCleanOrdersSheet, and click Run (safe — never wipes existing rows).
  2. When Updating Script Later: Never click Run again — just click Deploy → Manage deployments → Edit (pencil) → Version: New version → Deploy.
function backupOrdersTab() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Cakeflakes_Orders');
  if (!sheet || sheet.getLastRow() <= 1) return;
  let backup = ss.getSheetByName('Backup_Orders');
  if (!backup) backup = ss.insertSheet('Backup_Orders');
  backup.clear();
  sheet.getDataRange().copyTo(backup.getRange(1, 1));
}

function setupCleanOrdersSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Cakeflakes_Orders') || ss.getSheets()[0];
  sheet.setName('Cakeflakes_Orders');

  // Safeguard: If rows already exist, back them up and NEVER clear data
  if (sheet.getLastRow() > 0) {
    backupOrdersTab();
  }

  // Clean Row 1 Headers (15 Columns: A through O — 1-to-1 with Dashboard)
  const headers = [
    'Order_ID', 'Pickup_Date', 'Customer_Name', 'Phone', 'Event_Type',
    'Theme_Title', 'Tiers', 'Tier_Specs', 'Frosting_Colors', 'Notes',
    'Total_Price', 'Advance_Paid', 'Advance_Mode', 'Final_Paid', 'Phase'
  ];
  sheet.getRange(1, 1, 1, headers.length)
    .setValues([headers])
    .setFontWeight('bold')
    .setBackground('#9D174D')
    .setFontColor('#FFFFFF');
  sheet.setFrozenRows(1);

  // Format Pickup_Date column as plain YYYY-MM-DD text so dates never shift
  sheet.getRange('B2:B500').setNumberFormat('@STRING@');

  // Add Data Validation Dropdowns
  const tierRule = SpreadsheetApp.newDataValidation()
    .requireValueInList(['1 Tier', '2 Tier', '3 Tier', 'Sheet / Box'], true).build();
  const modeRule = SpreadsheetApp.newDataValidation()
    .requireValueInList(['Zelle', 'Credit Card', 'Cash'], true).build();
  const phaseRule = SpreadsheetApp.newDataValidation()
    .requireValueInList([
      'Order Received', 'Baking Finished', 'Designing Phase',
      'Ready for Pickup', 'Finally Delivered / Completed'
    ], true).build();

  sheet.getRange('G2:G500').setDataValidation(tierRule);
  sheet.getRange('M2:M500').setDataValidation(modeRule);
  sheet.getRange('O2:O500').setDataValidation(phaseRule);
}

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Cakeflakes_Orders') || ss.getSheets()[0];
  const tz = ss.getSpreadsheetTimeZone();
  const fmtDate = (val) => (val instanceof Date)
    ? Utilities.formatDate(val, tz, 'yyyy-MM-dd')
    : String(val || '').trim();

  const rows = sheet.getDataRange().getValues();
  const orders = rows.slice(1)
    .filter(r => r[2] && String(r[2]).trim() !== '')
    .map((r, i) => ({
      id: String(r[0] || `CF-${101 + i}`).trim(),
      pickupDate: fmtDate(r[1]),
      customerName: String(r[2] || '').trim(),
      phone: String(r[3] || '').trim(),
      eventType: String(r[4] || 'Birthday').trim(),
      themeTitle: String(r[5] || '').trim(),
      tiers: String(r[6] || '1 Tier').trim(),
      tierSpecs: String(r[7] || '').trim(),
      frostingColors: String(r[8] || '').trim(),
      notes: String(r[9] || '').trim(),
      totalPrice: parseFloat(r[10]) || 0,
      advancePaid: parseFloat(r[11]) || 0,
      advanceMode: String(r[12] || 'Zelle').trim(),
      finalPaid: parseFloat(r[13]) || 0,
      phase: String(r[14] || 'Order Received').trim()
    }));

  const jsonStr = JSON.stringify({ orders });
  const cb = e && e.parameter && e.parameter.callback;
  if (cb) {
    return ContentService.createTextOutput(`${cb}(${jsonStr})`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonStr)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('Cakeflakes_Orders') || ss.getSheets()[0];
    const o = payload.order;
    if (!o) return ContentService.createTextOutput(JSON.stringify({ status: 'no_order' }));

    const rows = sheet.getDataRange().getValues();
    const targetId = String(o.id || '').trim().toLowerCase();
    const targetCust = String(o.customerName || '').trim().toLowerCase();
    const targetDate = String(o.pickupDate || '').trim();

    let matchRow = -1;
    // 1. Primary match: exact Order_ID in Column A
    if (targetId) {
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0] || '').trim().toLowerCase() === targetId) {
          matchRow = i + 1;
          break;
        }
      }
    }
    // 2. Fallback match: Customer_Name + Pickup_Date (prevents duplicate if Order_ID was blank in sheet)
    if (matchRow === -1 && targetCust && targetDate) {
      for (let i = 1; i < rows.length; i++) {
        const rowCust = String(rows[i][2] || '').trim().toLowerCase();
        const rowDate = (rows[i][1] instanceof Date)
          ? Utilities.formatDate(rows[i][1], ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd')
          : String(rows[i][1] || '').trim();
        if (rowCust === targetCust && rowDate === targetDate) {
          matchRow = i + 1;
          break;
        }
      }
    }

    if (payload.action === 'delete') {
      if (matchRow > 1) sheet.deleteRow(matchRow);
      return ContentService.createTextOutput(JSON.stringify({ status: 'deleted' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const rowValues = [[
      o.id, o.pickupDate, o.customerName, o.phone || '', o.eventType || 'Birthday',
      o.themeTitle || '', o.tiers || '1 Tier', o.tierSpecs || '', o.frostingColors || '',
      o.notes || '', o.totalPrice || 0, o.advancePaid || 0, o.advanceMode || 'Zelle',
      o.finalPaid || 0, o.phase || 'Order Received'
    ]];

    if (matchRow > 1) {
      sheet.getRange(matchRow, 1, 1, 15).setValues(rowValues);
    } else {
      sheet.appendRow(rowValues[0]);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'saved', row: matchRow }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

Read-Only Script for "Bank reconcilation" (All Transactions Cols A–J, Row 5+)
  1. Open your Bank reconcilation Google Sheet → Extensions → Apps Script.
  2. Paste the script below, click Save, then click Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone).
  3. Copy the Web App URL and paste it into the box above.
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('All Transactions') || ss.getSheets()[0];
  const tz = ss.getSpreadsheetTimeZone();
  const fmtDate = (val) => (val instanceof Date)
    ? Utilities.formatDate(val, tz, 'yyyy-MM-dd')
    : String(val || '').trim();

  // Header is on Row 4; transactions start at Row 5 (Cols A through J)
  const rows = sheet.getDataRange().getValues();
  const transactions = rows.slice(4)
    .filter(r => r[1] || r[6] || r[7] || r[8])
    .map((r, i) => ({
      txNum: String(r[0] || (i + 1)).trim(),
      date: fmtDate(r[1]),
      year: String(r[2] || '').trim(),
      statementMonth: String(r[3] || '').trim(),
      section: String(r[4] || '').trim(),
      category: String(r[5] || '').trim(),
      party: String(r[6] || '').trim(),
      deposit: parseFloat(r[7]) || 0,
      withdrawal: parseFloat(r[8]) || 0,
      netImpact: parseFloat(r[9]) || ((parseFloat(r[7]) || 0) - (parseFloat(r[8]) || 0))
    }));

  const jsonStr = JSON.stringify({ transactions });
  const cb = e && e.parameter && e.parameter.callback;
  if (cb) {
    return ContentService.createTextOutput(`${cb}(${jsonStr})`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonStr)
    .setMimeType(ContentService.MimeType.JSON);
}
